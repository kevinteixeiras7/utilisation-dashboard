function convertOriginalReportToV2State(report, overtimePayload, capacityPayload) { const staffData = report.staffData || {}; const clientHours = report.clientHours || {}; const clientMeta = report.clientMeta || {}; const deptMap = {}; DEPT_ORDER.forEach(dept => { deptMap[dept] = { Art: dept, hours: 0, inv: 0, internal: 0, absence: 0, norm: 0, sale: 0, cost: 0, profit: 0, margin: 0, util: 0 }; }); Object.entries(staffData).forEach(([name, s]) => { if (!deptMap[s.dept]) return; deptMap[s.dept].hours += (s.cli || 0) + (s.int || 0) + (s.hol || 0) + (s.sck || 0); deptMap[s.dept].inv += s.bil || 0; deptMap[s.dept].internal += s.int || 0; deptMap[s.dept].absence += (s.hol || 0) + (s.sck || 0); deptMap[s.dept].norm += s.cap || 0; }); Object.values(deptMap).forEach(d => { d.util = d.norm ? (d.inv / d.norm) * 100 : 0; }); const people = Object.entries(staffData).map(([name, s]) => ({ EmpName: name, Art: s.dept, hours: (s.cli || 0) + (s.int || 0) + (s.hol || 0) + (s.sck || 0), inv: s.bil || 0, internal: s.int || 0, absence: (s.hol || 0) + (s.sck || 0), norm: s.cap || 0, util: s.cap ? ((s.cli || 0) / s.cap) * 100 : 0, billableUtil: s.cap ? ((s.bil || 0) / s.cap) * 100 : 0, sale: 0, cost: 0, profit: 0, margin: 0 })); const clients = Object.entries(clientHours).map(([name, hours]) => ({ CustomerName: name, hours, inv: hours, jobs: 0, sale: 0, cost: 0, profit: 0, margin: 0, ehr: 0, jobType: clientMeta[name]?.jobType || 'Other' })); const totalClientHours = Object.values(staffData).reduce((a, s) => a + (s.cli || 0), 0); const totalCapacity = Object.values(staffData).reduce((a, s) => a + (s.cap || 0), 0); const totalBillable = Object.values(staffData).reduce((a, s) => a + (s.bil || 0), 0); return { period: report.period, updatedAt: report.updatedAt, kpis: { totalHours: Object.values(staffData).reduce((a, s) => a + (s.cli || 0) + (s.int || 0) + (s.hol || 0) + (s.sck || 0), 0), billableHours: totalBillable, clientHours: totalClientHours, norm: totalCapacity, util: totalCapacity ? (totalClientHours / totalCapacity) * 100 : 0, clients: clients.length, jobs: 0, sale: 0, cost: 0, profit: 0, margin: 0, sets: 0, overtimeHours: overtimePayload?.otData ? Object.values(overtimePayload.otData).reduce((a, o) => a + (o.otHrs || 0), 0) : 0 }, dept: Object.values(deptMap), people, clients, jobs: [], sets: { byType: [], byPerson: [], byClient: [], total: 0 }, overtime: overtimePayload?.otData || {} }; }
